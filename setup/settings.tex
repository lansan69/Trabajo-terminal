% D. FICHAS DE REQUERIMIENTOS FUNCIONALES        (5.1.1 Requerimientos funcionales)
% ---------------------------------------------------------
% Se definen en el preámbulo porque babel (spanish) activa los
% caracteres < y > dentro del documento y rompe las columnas >{...}.

% \rfAncha{texto}
% Celda que abarca las columnas 2 a 4 de la tabla.
% Ancho: 1.00 + 0.60 + 1.45 = 3.05 (en unidades de \hsize),
% más los separadores (4\tabcolsep) y las líneas (2\arrayrulewidth)
% para que alinee con el resto de la tabla.
\newcommand{\rfAncha}[1]{%
  \multicolumn{3}{>{\hsize=\dimexpr3.05\hsize+4\tabcolsep+2\arrayrulewidth\relax\raggedright\arraybackslash}X|}{#1}}

% \tablaRF{Identificador}{Nombre}{Prioridad}{Requerimiento}
%         {Entrada}{Salida}{Descripción}{Precondición}{Postcondición}
%
% Genera la ficha de un requerimiento funcional con 4 columnas:
%   col 1 (.95): etiquetas, en negritas
%   col 2 (1.00): valores
%   col 3 (.60): etiquetas, en negritas
%   col 4 (1.45): valores
\newcommand{\tablaRF}[9]{%
  \par\noindent
  \begingroup
  \small                              % Mismo tamaño de letra que el resto de las tablas
  \setlength{\extrarowheight}{2pt}%   % Un poco de aire vertical en cada fila
  \begin{tabularx}{\linewidth}{%
      |>{\hsize=.95\hsize\raggedright\arraybackslash\bfseries}X%
      |>{\hsize=1.00\hsize\raggedright\arraybackslash}X%
      |>{\hsize=.60\hsize\raggedright\arraybackslash\bfseries}X%
      |>{\hsize=1.45\hsize\raggedright\arraybackslash}X|}
    \hline
    % Fila 1: requerimiento (celda ancha)
    Requerimiento & \rfAncha{#4} \\ \hline
    % Fila 2: identificador y nombre
    Identificador & #1 & Nombre & #2 \\ \cline{1-2}
    % Fila 3: prioridad (las dos últimas celdas quedan vacías)
    Prioridad de desarrollo & #3 & & \\ \hline
    % Fila 4: entrada y salida
    Entrada & #5 & Salida & #6 \\ \hline
    % Filas 5-7: texto largo en celda ancha
    Descripción & \rfAncha{#7} \\ \hline
    Precondición & \rfAncha{#8} \\ \hline
    Postcondición & \rfAncha{#9} \\ \hline
  \end{tabularx}%
  \endgroup
  \par\bigskip                         % Espacio antes de la siguiente tabla
}

% Empieza los mensajes 0.5 cm debajo de la cabecera más baja
\newcommand{\sdIniciar}{%
  \setcounter{sdNumero}{0}%
  \xdef\sdY{0}%
  \foreach \sdP in \sdLista {%
    \ifx\sdP\empty\else
      \path let \p1=(\sdP-base) in \pgfextra{%
        \pgfmathparse{min(\sdY, \y1/28.45274 - 0.5)}\xdef\sdY{\pgfmathresult}}; \fi}}

% Altura de un mensaje: 0.33 cm por renglón de texto más 0.25 cm de aire
% Cada renglón reserva 0.43 cm: \footnotesize en la clase de 12 pt usa un
% interlineado de 12 pt; con menos, el texto (fondo blanco) tapa la flecha
% anterior.
\newcommand{\sdMensaje}[4][1]{%
  \sdAvanzar{0.43*#1+0.15}\stepcounter{sdNumero}%
  \draw[sdLlamada] (#2 |- 0,\sdY) -- (#3 |- 0,\sdY)
    node[sdTexto, midway, above] {\arabic{sdNumero}. #4};}
\newcommand{\sdRespuesta}[4][1]{%
  \sdAvanzar{0.43*#1+0.15}\stepcounter{sdNumero}%
  \draw[sdVuelta] (#2 |- 0,\sdY) -- (#3 |- 0,\sdY)
    node[sdTexto, midway, above] {\arabic{sdNumero}. #4};}
\newcommand{\sdPropio}[3][1]{%
  \sdAvanzar{0.3}\stepcounter{sdNumero}%
  \draw[sdLlamada] (#2 |- 0,\sdY) -- ++(0.45,0) -- ++(0,-0.43*#1+0.05) -- ++(-0.45,0);
  \node[sdTexto, anchor=north west, align=left] at ($(#2 |- 0,\sdY)+(0.5,0.17)$) {\arabic{sdNumero}. #3};
  \sdAvanzar{0.43*#1}}
% Igual, pero el lazo y el texto van a la izquierda (para la última línea de vida)
\newcommand{\sdPropioIzq}[3][1]{%
  \sdAvanzar{0.3}\stepcounter{sdNumero}%
  \draw[sdLlamada] (#2 |- 0,\sdY) -- ++(-0.45,0) -- ++(0,-0.43*#1+0.05) -- ++(0.45,0);
  \node[sdTexto, anchor=north east, align=right] at ($(#2 |- 0,\sdY)+(-0.5,0.17)$) {\arabic{sdNumero}. #3};
  \sdAvanzar{0.43*#1}}

% Fragmentos: se guarda la altura de inicio, los extremos y el margen; el marco
% se dibuja al cerrar. [margen]: distancia del marco a las líneas de vida
% extremas (0.75 cm; usar uno menor en fragmentos anidados).
\newcommand{\sdAbrir}[6][0.75]{%
  \sdAvanzar{0.45}%
  \expandafter\xdef\csname sdYa@#2\endcsname{\sdY}%
  \expandafter\xdef\csname sdIz@#2\endcsname{#5}%
  \expandafter\xdef\csname sdDe@#2\endcsname{#6}%
  \expandafter\xdef\csname sdMa@#2\endcsname{#1}%
  \node[sdEtiquetaMarco, anchor=north west] (sdTab-#2) at ($(#5 |- 0,\sdY)+(-#1,0)$) {#3};
  \draw[sdMarco] (sdTab-#2.south west) -| ([xshift=4pt]sdTab-#2.east) -- ([xshift=4pt, yshift=-4pt]sdTab-#2.north east) |- (sdTab-#2.north west);
  \node[sdGuarda, anchor=west] at ([xshift=6pt]sdTab-#2.east) {#4};
  \sdAvanzar{0.3}}
\newcommand{\sdSino}[2]{%
  \sdAvanzar{0.3}%
  \draw[sdMarco, dashed] ($(\csname sdIz@#1\endcsname |- 0,\sdY)+(-\csname sdMa@#1\endcsname,0)$) -- ($(\csname sdDe@#1\endcsname |- 0,\sdY)+(\csname sdMa@#1\endcsname,0)$);
  \node[sdGuarda, anchor=north west] at ($(\csname sdIz@#1\endcsname |- 0,\sdY)+(-\csname sdMa@#1\endcsname+0.05,0)$) {#2};
  \sdAvanzar{0.3}}
\newcommand{\sdCerrar}[1]{%
  \sdAvanzar{0.3}%
  \draw[sdMarco] ($(\csname sdIz@#1\endcsname |- 0,\csname sdYa@#1\endcsname)+(-\csname sdMa@#1\endcsname,0)$)
    rectangle ($(\csname sdDe@#1\endcsname |- 0,\sdY)+(\csname sdMa@#1\endcsname,0)$);}
