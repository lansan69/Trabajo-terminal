% Estilos y macros: setup/settings.tex, sección G3.
% La lógica de ruteo vive en una función de la base de datos (pgRouting)
% que la aplicación invoca como RPC a través de la API REST.
% =========================================================
\begin{tikzpicture}

  % ---------------- Participantes ----------------
  \sdActor{usr}{0}{Peatón\\usuario}
  \sdContenedor{app}{3.2}{Aplicación móvil}{Flutter}
  \sdContenedor{api}{6.8}{API REST}{Supabase PostgREST}
  \sdContenedor{bd}{11.4}{Base de datos}{PostGIS, pgRouting}
  \sdIniciar

  \sdMensaje{usr}{app}{solicita las rutas}

  \sdAbrir{red}{alt}{[sin conexión a internet]}{usr}{bd}
    \sdMensaje[2]{app}{usr}{avisa que no\\hay conexión}
  \sdSino{red}{[con conexión]}
    \sdMensaje[3]{app}{api}{solicita las rutas entre\\el origen y el destino\\a la hora actual}
    \sdMensaje{api}{bd}{ejecuta el cálculo de rutas}
    \sdPropioIzq[2]{bd}{valida el origen y el destino\\en el área de servicio}
    \sdPropioIzq[2]{bd}{toma los pesos de la\\franja horaria}
    \sdPropioIzq[2]{bd}{calcula las rutas segura,\\rápida y balanceada}

    \sdAbrir[0.55]{resultado}{alt}{[origen o destino fuera del área]}{app}{bd}
      \sdRespuesta{bd}{api}{error de cobertura}
      \sdRespuesta{api}{app}{error de cobertura}
      \sdMensaje[2]{app}{usr}{indica la restricción\\de cobertura}
    \sdSino{resultado}{[no existe ruta peatonal]}
      \sdRespuesta{bd}{api}{sin resultados}
      \sdRespuesta{api}{app}{sin resultados}
      \sdMensaje[2]{app}{usr}{avisa que no\\existe ruta}
    \sdSino{resultado}{[rutas encontradas]}
      \sdRespuesta[2]{bd}{api}{rutas con distancia y\\riesgo por tramo}
      \sdRespuesta[2]{api}{app}{rutas en GeoJSON\\y resumen}
      \sdPropio[2]{app}{calcula el tiempo a 4.5 km/h\\y marca las rutas iguales}
      \sdMensaje[3]{app}{usr}{muestra las rutas con\\distancia, tiempo\\y riesgo}
    \sdCerrar{resultado}
  \sdCerrar{red}

  \sdTerminar
\end{tikzpicture}
